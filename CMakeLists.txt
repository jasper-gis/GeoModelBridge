cmake_minimum_required(VERSION 3.21)
project(GeoModelBridge VERSION 0.1.6 LANGUAGES C CXX)

option(GMB_BUILD_TESTS "Build correctness tests" ON)
option(GMB_BUILD_NATIVE "Build the native FileGDB writer (requires the platform SDK)" OFF)
option(GMB_STATIC_RUNTIME "Link MinGW runtime statically for a portable CLI" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(gmb_ufbx STATIC third_party/ufbx/ufbx.c)
target_include_directories(gmb_ufbx PUBLIC third_party/ufbx)
if(UNIX)
  target_link_libraries(gmb_ufbx PUBLIC m)
endif()
add_library(GeoModelBridge.Core STATIC src/core.cpp src/bundle.cpp src/fbx_reader.cpp src/fixtures.cpp)
add_library(gmb::core ALIAS GeoModelBridge.Core)
target_include_directories(GeoModelBridge.Core PUBLIC include PRIVATE third_party)
target_link_libraries(GeoModelBridge.Core PRIVATE gmb_ufbx)
add_executable(geomodelbridge src/main.cpp)
target_include_directories(geomodelbridge PRIVATE third_party)
target_link_libraries(geomodelbridge PRIVATE gmb::core)
if(WIN32)
  target_link_libraries(geomodelbridge PRIVATE shell32)
endif()
foreach(target GeoModelBridge.Core geomodelbridge)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(MINGW AND GMB_STATIC_RUNTIME)
  target_link_options(geomodelbridge PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING AND GMB_BUILD_TESTS)
  add_executable(gmb_core_tests tests/core_tests.cpp)
  target_link_libraries(gmb_core_tests PRIVATE gmb::core)
  target_include_directories(gmb_core_tests PRIVATE third_party)
  if(MINGW AND GMB_STATIC_RUNTIME)
    target_link_options(gmb_core_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core COMMAND gmb_core_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py")
    add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py" "$<TARGET_FILE:geomodelbridge>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME reader_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reader_integration_test.py" "$<TARGET_FILE:geomodelbridge>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME binary_fbx COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/binary_fbx_test.py" "$<TARGET_FILE:geomodelbridge>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME profile_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/profile_integration_test.py" "$<TARGET_FILE:geomodelbridge>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  endif()
endif()
install(TARGETS geomodelbridge RUNTIME DESTINATION bin)
install(FILES tests/fixtures/textured_quad.fbx tests/fixtures/checker.png DESTINATION bin/demo)
if(GMB_BUILD_NATIVE)
  add_subdirectory(backends/native-filegdb)
endif()
install(DIRECTORY docs DESTINATION . PATTERN "*.lock" EXCLUDE)
install(FILES README.md VERSION CHANGELOG.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/ufbx/LICENSE DESTINATION licenses/ufbx)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses/nlohmann)
